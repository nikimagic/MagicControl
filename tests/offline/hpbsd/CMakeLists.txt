cmake_minimum_required(VERSION 3.25)
project(HpbsdOfflineChecks LANGUAGES CXX)
get_filename_component(MC_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
add_executable(hpbsd-offline Checks.cpp)
target_compile_features(hpbsd-offline PRIVATE cxx_std_20)
target_include_directories(hpbsd-offline PRIVATE "${MC_ROOT}" "${MC_ROOT}/src/core/include")
if(MSVC)
 target_compile_options(hpbsd-offline PRIVATE /utf-8 /W4)
endif()
