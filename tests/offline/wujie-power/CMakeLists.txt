project(WujiePowerReadbackChecks LANGUAGES CXX)
get_filename_component(MC_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
add_executable(wujie-power-offline Checks.cpp)
