cmake_minimum_required(VERSION 3.25)
project(MagicControlControlRuntimeOffline LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
get_filename_component(MC_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(runtime_offline test_runtime.cpp "${MC_ROOT}/src/core/src/ActionLog.cpp")
target_compile_features(runtime_offline PRIVATE cxx_std_20)
target_include_directories(runtime_offline PRIVATE "${MC_ROOT}/src/core/include")
target_link_libraries(runtime_offline PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(runtime_offline PRIVATE /utf-8 /W4)
endif()
add_test(NAME finite_runtime_isolation COMMAND runtime_offline)
add_test(NAME power_source_boundaries COMMAND "${CMAKE_COMMAND}" "-DSOURCE_ROOT=${MC_ROOT}" -P "${MC_ROOT}/tests/architecture_audit.cmake")
set_tests_properties(finite_runtime_isolation PROPERTIES TIMEOUT 25)
# The full source audit reads the shared Windows tree from WSL/NTFS as well.
# This is an I/O budget; runtime synchronization still has its own tight bounds.
set_tests_properties(power_source_boundaries PROPERTIES TIMEOUT 90)
add_executable(native_lighting_offline test_native_lighting.cpp)
target_compile_features(native_lighting_offline PRIVATE cxx_std_20)
target_include_directories(native_lighting_offline PRIVATE "${MC_ROOT}/src/core/include")
if(MSVC)
    target_compile_options(native_lighting_offline PRIVATE /utf-8 /W4)
endif()
add_test(NAME native_lighting_shared_entry COMMAND native_lighting_offline)
set_tests_properties(native_lighting_shared_entry PROPERTIES TIMEOUT 25)
add_executable(native_battery_offline test_native_battery.cpp)
target_compile_features(native_battery_offline PRIVATE cxx_std_20)
target_include_directories(native_battery_offline PRIVATE "${MC_ROOT}/src/core/include")
if(MSVC)
    target_compile_options(native_battery_offline PRIVATE /utf-8 /W4)
endif()
add_test(NAME native_battery_shared_entry COMMAND native_battery_offline)
set_tests_properties(native_battery_shared_entry PROPERTIES TIMEOUT 25)
add_executable(battery_contract_offline "${MC_ROOT}/tests/offline/test_battery.cpp")
target_compile_features(battery_contract_offline PRIVATE cxx_std_20)
target_include_directories(battery_contract_offline PRIVATE "${MC_ROOT}" "${MC_ROOT}/src/core/include")
target_compile_definitions(battery_contract_offline PRIVATE MC_FIXTURES="${MC_ROOT}/tests/offline/fixtures")
if(MSVC)
    target_compile_options(battery_contract_offline PRIVATE /utf-8 /W4)
endif()
add_test(NAME battery_protocol_contract COMMAND battery_contract_offline)
set_tests_properties(battery_protocol_contract PROPERTIES TIMEOUT 45)
add_executable(fan_request_offline test_fan_request.cpp)
target_compile_features(fan_request_offline PRIVATE cxx_std_20)
target_include_directories(fan_request_offline PRIVATE "${MC_ROOT}/src/core/include")
if(MSVC)
    target_compile_options(fan_request_offline PRIVATE /utf-8 /W4)
endif()
add_test(NAME fan_request_contract COMMAND fan_request_offline)
set_tests_properties(fan_request_contract PROPERTIES TIMEOUT 25)
if(WIN32)
    set(MC_LIB "${MC_ROOT}/build/windows/lib/Release")
    add_executable(power_entry_offline test_power_entry.cpp "${MC_ROOT}/tests/test_board_fixture.cpp")
    add_executable(lighting_entry_offline test_lighting_windows.cpp "${MC_ROOT}/tests/test_board_fixture.cpp")
    add_executable(fan_entry_offline test_fan_entry_windows.cpp "${MC_ROOT}/tests/test_board_fixture.cpp")
    # Compile the adapted existing fixture without running its broader suite.
    add_executable(settings_queue_compile "${MC_ROOT}/tests/test_settings_queue_windows.cpp")
    foreach(target power_entry_offline lighting_entry_offline native_battery_offline fan_entry_offline settings_queue_compile)
        target_compile_features(${target} PRIVATE cxx_std_20)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
        target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
        target_include_directories(${target} PRIVATE "${MC_ROOT}/src/core/include" "${MC_ROOT}/build/windows/generated")
        target_link_libraries(${target} PRIVATE
            "${MC_LIB}/magiccontrol_platform.lib" "${MC_LIB}/magiccontrol_core.lib"
            "${MC_LIB}/magiccontrol_storage.lib" "${MC_LIB}/magiccontrol_runtime.lib"
            advapi32 ole32 oleaut32 wbemuuid setupapi cfgmgr32 tbs winhttp windowsapp
            user32 powrprof uuid bcrypt dxgi version wintrust crypt32 shell32)
    endforeach()
    target_compile_definitions(settings_queue_compile PRIVATE MAGICCONTROL_TEST_ROOT="${CMAKE_BINARY_DIR}")
    add_test(NAME native_power_single_entry COMMAND power_entry_offline)
    set_tests_properties(native_power_single_entry PROPERTIES TIMEOUT 25)
    add_test(NAME lighting_singlezone_memory COMMAND lighting_entry_offline "${CMAKE_CURRENT_BINARY_DIR}/lighting-fixture.ini")
    set_tests_properties(lighting_singlezone_memory PROPERTIES TIMEOUT 45)
    add_test(NAME fan_shared_executor_memory COMMAND fan_entry_offline)
    set_tests_properties(fan_shared_executor_memory PROPERTIES TIMEOUT 75)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # Type-check the adapted old Linux fixture without running its broader suite.
    add_library(hlyl_policy_compile OBJECT "${MC_ROOT}/tests/test_hlyl_battery_policy_linux.cpp")
    target_compile_features(hlyl_policy_compile PRIVATE cxx_std_20)
    target_include_directories(hlyl_policy_compile PRIVATE "${MC_ROOT}/src/core/include")
endif()
