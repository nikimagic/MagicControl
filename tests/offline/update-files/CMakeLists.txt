cmake_minimum_required(VERSION 3.25)
project(UpdateReplaceFileChecks LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Explicit Windows file replacement check")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
get_filename_component(MC_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
add_executable(update-file-checks Checks.cpp)
target_compile_features(update-file-checks PRIVATE cxx_std_20)
target_compile_definitions(update-file-checks PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN MC_FILES_ROOT="${CMAKE_CURRENT_BINARY_DIR}/fixtures")
target_include_directories(update-file-checks PRIVATE "${MC_ROOT}/src/update")
target_link_libraries(update-file-checks PRIVATE "${MC_ROOT}/build/windows/lib/Release/magiccontrol_update.lib" "${MC_ROOT}/build/windows/lib/Release/magiccontrol_runtime.lib" bcrypt wintrust crypt32 shell32 advapi32)
