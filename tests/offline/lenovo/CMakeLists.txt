cmake_minimum_required(VERSION 3.25)
project(LegionProtocolOffline LANGUAGES CXX)
enable_testing()
get_filename_component(MC_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
add_executable(legion_protocol Checks.cpp)
target_sources(legion_protocol PRIVATE "${MC_ROOT}/src/devices/lenovo/legion-kwcn/Profile.cpp")
target_compile_features(legion_protocol PRIVATE cxx_std_20)
target_include_directories(legion_protocol PRIVATE "${MC_ROOT}/src/core/include" "${MC_ROOT}")
if(MSVC)
    set_property(TARGET legion_protocol PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    target_compile_options(legion_protocol PRIVATE /utf-8 /W4)
endif()
add_test(NAME legion_wmi_curve_contract COMMAND legion_protocol)
set_tests_properties(legion_wmi_curve_contract PROPERTIES TIMEOUT 30)
add_executable(legion_fan_fault_matrix FanFaultMatrix.cpp)
target_compile_features(legion_fan_fault_matrix PRIVATE cxx_std_20)
target_include_directories(legion_fan_fault_matrix PRIVATE "${MC_ROOT}/src/core/include" "${MC_ROOT}")
if(MSVC)
    target_compile_options(legion_fan_fault_matrix PRIVATE /utf-8 /W4)
endif()
add_test(NAME legion_fan_fault_matrix COMMAND legion_fan_fault_matrix)
set_tests_properties(legion_fan_fault_matrix PROPERTIES TIMEOUT 30 LABELS "offline;lenovo;fans")
add_executable(legion_power_observation ../test_lenovo_power_observation.cpp)
target_compile_features(legion_power_observation PRIVATE cxx_std_20)
target_include_directories(legion_power_observation PRIVATE "${MC_ROOT}/src/core/include" "${MC_ROOT}")
if(MSVC)
    target_compile_options(legion_power_observation PRIVATE /utf-8 /W4)
endif()
add_test(NAME legion_power_observation_offline COMMAND legion_power_observation)
set_tests_properties(legion_power_observation_offline PROPERTIES TIMEOUT 15 LABELS "offline;lenovo;power")
if(WIN32)
    add_executable(legion_runtime Runtime.cpp "${MC_ROOT}/tests/test_board_fixture.cpp")
    target_compile_features(legion_runtime PRIVATE cxx_std_20)
    set_property(TARGET legion_runtime PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    target_compile_options(legion_runtime PRIVATE /utf-8 /W4)
    target_compile_definitions(legion_runtime PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(legion_runtime PRIVATE "${MC_ROOT}" "${MC_ROOT}/src/core/include" "${MC_ROOT}/build/windows/generated")
    set(MC_LIB "${MC_ROOT}/build/windows/lib/Release")
    target_link_libraries(legion_runtime PRIVATE
        "${MC_LIB}/magiccontrol_platform.lib" "${MC_LIB}/magiccontrol_core.lib"
        "${MC_LIB}/magiccontrol_storage.lib" "${MC_LIB}/magiccontrol_runtime.lib"
        advapi32 ole32 oleaut32 wbemuuid setupapi cfgmgr32 tbs winhttp windowsapp
        user32 powrprof uuid bcrypt dxgi version wintrust crypt32 shell32)
    add_custom_command(TARGET legion_runtime POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:legion_runtime>/settings"
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${MC_ROOT}/packaging/fan-curves.csv" "$<TARGET_FILE_DIR:legion_runtime>/settings/fan-curves.csv")
    add_test(NAME legion_shared_custom_and_restore COMMAND legion_runtime)
    set_tests_properties(legion_shared_custom_and_restore PROPERTIES TIMEOUT 30)
endif()
