cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SensorProbe VERSION 1.0 LANGUAGES CXX)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(sensor-probe src/main.cpp src/Nvidia.cpp)
target_compile_features(sensor-probe PRIVATE cxx_std_20)
target_include_directories(sensor-probe SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/Intel" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/NVIDIA")
if(WIN32)
    target_include_directories(sensor-probe SYSTEM PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/vendor/AMD/ADL/include"
        "${CMAKE_CURRENT_SOURCE_DIR}/vendor/AMD/ADLX/include")
    target_sources(sensor-probe PRIVATE src/Windows.cpp src/AmdWindows.cpp src/PawnWindows.cpp src/IntelWindows.cpp src/StorageWindows.cpp src/CommandWindows.cpp SensorProbe.manifest)
    target_link_libraries(sensor-probe PRIVATE advapi32 ole32 oleaut32 wbemuuid setupapi pdh user32 powrprof uuid)
    target_compile_definitions(sensor-probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_compile_options(sensor-probe PRIVATE /W4 /utf-8 /MP)
    target_link_options(sensor-probe PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/SensorProbe.manifest" "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
    set(probe_platform windows)
    add_executable(ec-bit-test src/EcBitTestWindows.cpp EcBitTest.manifest)
    target_compile_features(ec-bit-test PRIVATE cxx_std_20)
    target_compile_definitions(ec-bit-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_compile_options(ec-bit-test PRIVATE /W4 /utf-8)
    target_link_libraries(ec-bit-test PRIVATE advapi32 user32 ole32 oleaut32 wbemuuid)
    target_link_options(ec-bit-test PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/EcBitTest.manifest" "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
    set_target_properties(ec-bit-test PROPERTIES OUTPUT_NAME EcBitTest)
    add_executable(ec-bit-test-selfcheck src/EcBitTestWindows.cpp)
    target_compile_features(ec-bit-test-selfcheck PRIVATE cxx_std_20)
    target_compile_definitions(ec-bit-test-selfcheck PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS EC_BIT_SELF_TEST_ONLY)
    target_compile_options(ec-bit-test-selfcheck PRIVATE /W4 /utf-8)
    target_link_libraries(ec-bit-test-selfcheck PRIVATE advapi32 user32 ole32 oleaut32 wbemuuid)
    target_link_options(ec-bit-test-selfcheck PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
    set_target_properties(ec-bit-test-selfcheck PROPERTIES OUTPUT_NAME EcBitTestSelfCheck)
    add_custom_command(TARGET ec-bit-test POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:ec-bit-test>" "${CMAKE_CURRENT_SOURCE_DIR}/windows/EcBitTest.exe"
        VERBATIM)
else()
    target_sources(sensor-probe PRIVATE src/Linux.cpp src/CommandLinux.cpp)
    target_link_libraries(sensor-probe PRIVATE dl pthread)
    target_compile_options(sensor-probe PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(sensor-probe PRIVATE -static-libstdc++ -static-libgcc)
    set(probe_platform linux)
endif()
set_target_properties(sensor-probe PROPERTIES OUTPUT_NAME SensorProbe)
add_custom_command(TARGET sensor-probe POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_SOURCE_DIR}/${probe_platform}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:sensor-probe>" "${CMAKE_CURRENT_SOURCE_DIR}/${probe_platform}/$<TARGET_FILE_NAME:sensor-probe>"
    VERBATIM)
if(WIN32)
    add_custom_command(TARGET sensor-probe POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/runtime/PawnIO" "${CMAKE_CURRENT_SOURCE_DIR}/windows/PawnIO" VERBATIM)
endif()
