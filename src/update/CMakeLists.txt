add_library(magiccontrol_update STATIC AppUpdate.cpp Download.cpp)
target_include_directories(magiccontrol_update PUBLIC "${PROJECT_SOURCE_DIR}/src/core/include" "${CMAKE_BINARY_DIR}/generated")
add_dependencies(magiccontrol_update magiccontrol-build-version)
target_compile_features(magiccontrol_update PUBLIC cxx_std_20)
target_link_libraries(magiccontrol_update PUBLIC magiccontrol_core Threads::Threads)
if(WIN32)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/ApplyUpdateWindows.ps1")
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/ApplyUpdateWindows.ps1" MC_UPDATE_SCRIPT)
    configure_file(UpdateScript.h.in "${CMAKE_CURRENT_BINARY_DIR}/UpdateScript.h" @ONLY)
    target_include_directories(magiccontrol_update PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_sources(magiccontrol_update PRIVATE PlatformWindows.cpp PawnIoInstallWindows.cpp)
    target_link_libraries(magiccontrol_update PRIVATE magiccontrol_runtime bcrypt wintrust crypt32 shell32 advapi32)
    target_compile_definitions(magiccontrol_update PRIVATE NOMINMAX)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(UPDATE_GLIB REQUIRED IMPORTED_TARGET glib-2.0)
    target_sources(magiccontrol_update PRIVATE PlatformLinux.cpp)
    target_link_libraries(magiccontrol_update PRIVATE PkgConfig::UPDATE_GLIB)
endif()
