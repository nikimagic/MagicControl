cmake_minimum_required(VERSION 3.24)
project(nvflash_offline_checks LANGUAGES CXX)
get_filename_component(MC_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(nvflash_offline_checks checks.cpp)
target_include_directories(nvflash_offline_checks PRIVATE "${MC_ROOT}/src/platform/common" "${MC_ROOT}/src/core/include")
target_compile_features(nvflash_offline_checks PRIVATE cxx_std_20)
# No MagicControl core/platform, NVML, vendor payload or hardware API linked.
add_executable(collector_simulation simulation.cpp "${MC_ROOT}/src/platform/common/VbiosRomDump.cpp")
target_include_directories(collector_simulation PRIVATE "${MC_ROOT}/src/platform/common" "${MC_ROOT}/src/core/include" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_features(collector_simulation PRIVATE cxx_std_20)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated/MagicControl")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/MagicControl/Version.h" "#pragma once\n#define MAGICCONTROL_VERSION_STRING \"0.1.196-offline-simulation\"\n")
if(WIN32)
  target_compile_options(collector_simulation PRIVATE /utf-8)
  target_link_libraries(collector_simulation PRIVATE bcrypt)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
  target_link_libraries(collector_simulation PRIVATE PkgConfig::GLIB)
endif()
