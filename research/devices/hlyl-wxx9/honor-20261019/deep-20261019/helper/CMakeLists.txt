cmake_minimum_required(VERSION 3.25)
project(HonorDiagnostics LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
add_executable(HonorDiagnostics main.cpp)
target_compile_features(HonorDiagnostics PRIVATE cxx_std_20)
target_compile_definitions(HonorDiagnostics PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(HonorDiagnostics PRIVATE /utf-8 /EHsc)
target_include_directories(HonorDiagnostics PRIVATE "${REPO}" "${REPO}/src/core/include")
target_link_libraries(HonorDiagnostics PRIVATE "${REPO}/build/windows/lib/Release/magiccontrol_platform.lib" "${REPO}/build/windows/lib/Release/magiccontrol_core.lib" advapi32 ole32 oleaut32 wbemuuid setupapi cfgmgr32 tbs winhttp windowsapp user32 powrprof uuid bcrypt dxgi shell32 gdi32)
